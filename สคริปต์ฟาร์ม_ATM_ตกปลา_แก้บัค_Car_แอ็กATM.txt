local function refreshCharacter()
    local player = game:GetService("Players").LocalPlayer
    local character = player.Character or player.CharacterAdded:Wait()
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")
    return player, character, humanoid, root
end

local function getPlayerGui()
    local player = game:GetService("Players").LocalPlayer
    return player:FindFirstChildOfClass("PlayerGui")
end

local fishingRunId = 0

local function isRunning(runId)
    return Config.AutoFarmFishing and runId == fishingRunId
end

local function waitUntil(condition, timeout, interval)
    timeout = timeout or 5
    interval = interval or 0.1
    local started = os.clock()
    while os.clock() - started < timeout do
        if condition() then return true end
        task.wait(interval)
    end
    return condition()
end

-- 4) Inventory helpers
-- ============================================================
local function getHolding()
    local gui = getPlayerGui()
    local items = gui and gui:FindFirstChild("Items")
    local holder = items and items:FindFirstChild("ItemsHolder")
    return holder and holder:FindFirstChild("ItemsScrollingFrame")
end

local function findInventoryItem(itemName)
    local holding = getHolding()
    if not holding then return nil end

    for _, item in ipairs(holding:GetChildren()) do
        if item.Name ~= "Folder" and item.Name ~= "UIGridLayout" and item.Name ~= "ItemTemplate" then
            local label = item:FindFirstChild("ItemName")
            if label and string.lower(tostring(label.Text)) == string.lower(tostring(itemName)) then
                return item
            end
        end
    end
    return nil
end

local function CountFish()
    local count = 0
    local holding = getHolding()
    if not holding then return 0 end

    for _, item in ipairs(holding:GetChildren()) do
        if item:GetAttribute("ItemType") == "fish" then
            count += 1
        end
    end
    return count
end

local function getBestFishingItem(list, requested)
    if requested and requested ~= "Smart Select" then
        return requested
    end

    local level = 1
    pcall(function() level = tonumber(Sf:GetLevel()) or 1 end)

    local bestName = list[#list].Name
    local bestLevel = -1
    for _, item in ipairs(list) do
        if item.Level <= level and item.Level >= bestLevel then
            bestName = item.Name
            bestLevel = item.Level
        end
    end
    return bestName
end

local function getFishingVehicle()
    local vehicles = workspace:FindFirstChild("Vehicles")
    if not vehicles then return nil end

    for _, vehicle in ipairs(vehicles:GetChildren()) do
        local owner = vehicle:GetAttribute("OwnerUserId")
        local okType = false
        pcall(function()
            okType = Sf:IsVehicleType(vehicle, Config.FishingVehicleType)
        end)
        if tostring(owner) == tostring(UserId) and okType and vehicle.PrimaryPart then
            return vehicle
        end
    end
    return nil
end

local function ensureFishingVehicle(runId)
    local vehicle = getFishingVehicle()
    if vehicle then return vehicle end
    if not isRunning(runId) then return nil end

    -- พยายามเรียกรถที่มีอยู่ก่อน
    pcall(function() Sf:PullVehicle() end)
    task.wait(0.35)
    vehicle = getFishingVehicle()
    if vehicle then return vehicle end

    -- ถ้าเลือก Bike และไม่มีรถ ให้ซื้อ/สร้าง bike crate ตามระบบเดิม
    if Config.FishingVehicleType == "Bike" then
        pcall(function() Sf:BuyBikeCrate() end)
        task.wait(0.8)
    else
        local vehicleName
        pcall(function() vehicleName = Sf:GetCarFromType(Config.FishingVehicleType) end)
        if vehicleName then
            local info
            pcall(function() info = Sf:GetInfo(vehicleName) end)
            if info and info[2] and not info[4] and not info[5] then
                pcall(function()
                    Net.get("toggle_equip_item", tostring(info[3]))
                end)
                task.wait(0.8)
            end
        end
    end

    vehicle = getFishingVehicle()
    return vehicle
end

local function sitInFishingVehicle(vehicle, runId)
    if not vehicle or not vehicle.PrimaryPart or not isRunning(runId) then return false end

    local _, character, humanoid, root = refreshCharacter()
    if humanoid and humanoid.Sit then return true end

    pcall(function() Sf:PullVehicle() end)
    task.wait(0.25)

    -- พยายามจากระยะปกติก่อน
    for _ = 1, 3 do
        if not isRunning(runId) then return false end
        if Sf:EnterVehicle(vehicle, 8) then break end
        task.wait(0.25)
    end

    -- ถ้ายังไม่ขึ้นรถ ให้ย้ายตัวละครไปที่รถก่อน แล้วลองใหม่
    if humanoid and not humanoid.Sit and root then
        pcall(function()
            root.CFrame = CFrame.new(vehicle.PrimaryPart.Position + Vector3.new(0, 3, 0))
        end)
        task.wait(0.2)
        pcall(function() Sf:EnterVehicle(vehicle, 8) end)
    end

    return waitUntil(function()
        local _, _, h = refreshCharacter()
        return h and h.Sit
    end, 2, 0.05)
end

local function stopVehicle(vehicle)
    if not vehicle then return end
    for _, part in ipairs(vehicle:GetDescendants()) do
        if part:IsA("BasePart") then
            part.AssemblyLinearVelocity = Vector3.zero
            part.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

-- ============================================================
-- 6) Forced underground vehicle movement
-- ============================================================
local function driveUnderground(vehicle, destination, runId, stopDist, onArrive)
    if not vehicle or not vehicle.PrimaryPart then return false end
    if not isRunning(runId) then return false end

    if not sitInFishingVehicle(vehicle, runId) then
        return false
    end

    local depth = math.clamp(tonumber(Config.FarmUndergroundDepth) or 55, 10, 150)
    local speed = math.max(5, tonumber(Config.FishingDrivingSpeed) or 60)
    local undergroundY = destination.Y - depth

    -- ลงใต้ดินก่อนเริ่มวิ่งแนวราบ
    local pivot = vehicle:GetPivot()
    local rotation = pivot.Rotation
    local start = pivot.Position
    local undergroundStart = Vector3.new(start.X, undergroundY, start.Z)
    vehicle:PivotTo(CFrame.new(undergroundStart) * rotation)
    stopVehicle(vehicle)
    task.wait(0.08)

    -- เคลื่อนที่เฉพาะแกน X/Z ใต้ดิน ไม่ใช้ Pathfinding ที่อาจพารถกลับขึ้นพื้น
    local target = Vector3.new(destination.X, undergroundY, destination.Z)
    local current = vehicle:GetPivot().Position
    local flatDelta = Vector3.new(target.X - current.X, 0, target.Z - current.Z)
    local distance = flatDelta.Magnitude
    local direction = distance > 0.01 and flatDelta.Unit or Vector3.zero
    local travelled = 0

    while isRunning(runId) and vehicle and vehicle.PrimaryPart do
        task.wait()
        if not sitInFishingVehicle(vehicle, runId) then
            return false
        end

        local dt = 1 / 60
        travelled = math.min(distance, travelled + speed * dt)
        local pos = current + direction * travelled
        pos = Vector3.new(pos.X, undergroundY, pos.Z)

        local lookDir = direction.Magnitude > 0 and direction or Vector3.new(0, 0, -1)
        local cf = CFrame.lookAt(pos, pos + lookDir)

        if Config.FishingWheelieEnabled and (not Config.FishingWheelieOnlyBike or Config.FishingVehicleType == "Bike") then
            cf *= CFrame.Angles(math.rad(math.clamp(Config.FishingWheelieAngle or 0, 0, 35)), 0, 0)
        end

        vehicle:PivotTo(cf)
        stopVehicle(vehicle)

        local remaining = (vehicle.PrimaryPart.Position - target).Magnitude
        if travelled >= distance or (stopDist and remaining <= stopDist) then
            break
        end
    end

    if not isRunning(runId) then
        return false
    end

    -- ขึ้นพื้นตรงปลายทางเท่านั้น
    local finalPivot = vehicle:GetPivot()
    local finalRotation = finalPivot.Rotation
    vehicle:PivotTo(CFrame.new(destination) * finalRotation)
    stopVehicle(vehicle)
    task.wait(0.08)

    if onArrive then
        pcall(onArrive)
    end
    return true
end

local SHOP_POS = Vector3.new(-136.406662, 254.851074, 153.555588)
local ATM_POS = Vector3.new(179.200882, 255.290543, -236.355896)
local FISHING_SPOT = Vector3.new(184.753616, 254.942337, -751.416321)
local CAST_POS = Vector3.new(183.10549926757812, 249.84375, -761.474365234375)
local SELL_PARK_POS = Vector3.new(-299.4, 254.8, 434.4)

local function getSelectedEquipment()
    local rodName = getBestFishingItem(FishingRods, Config.FishingRod)
    local baitName = getBestFishingItem(FishingBaits, Config.FishingWormtec)
    local rod = findInventoryItem(rodName)
    local bait = findInventoryItem(baitName)
    return rodName, baitName, rod, bait
end

local function getPrice(list, name)
    for _, item in ipairs(list) do
        if item.Name == name then return item.Price end
    end
    return 0
end

local function travelToWithVehicle(vehicle, pos, runId)
    if not vehicle then return false end
    return driveUnderground(vehicle, pos, runId, 5, function()
        local _, character, humanoid = refreshCharacter()
        if humanoid and humanoid.Sit then
            pcall(function() Sf:ExitVehicle(5) end)
        end
    end)
end

local function ensureMoneyForEquipment(vehicle, rodName, baitName, runId, needRod, needBait)
    local required = 0
    if needRod then required += getPrice(FishingRods, rodName) end
    if needBait then required += getPrice(FishingBaits, baitName) * math.clamp(math.floor(tonumber(Config.FishingBaitAmount) or 15), 1, 50) end
    if required <= 0 then return true end

    local money = 0
    pcall(function() money = tonumber(Sf:GetMoney()) or 0 end)
    if money >= required then return true end

    if not travelToWithVehicle(vehicle, ATM_POS, runId) then return false end

    -- ใช้ระบบโอนเงินเดิมของสคริปต์ หากเงินสดไม่พอ
    local missing = required - money
    if missing > 0 then
        local ok = pcall(function()
            Net.get("transfer_funds", "bank", "hand", missing)
        end)
        if not ok then return false end
        task.wait(0.35)
    end

    return true
end

local function buyMissingEquipment(vehicle, rodName, baitName, runId)
    local _, _, rodItem, baitItem = getSelectedEquipment()
    local baitAmount = math.clamp(math.floor(tonumber(Config.FishingBaitAmount) or 15), 1, 50)

    if rodItem and baitItem then return true end

    if not ensureMoneyForEquipment(vehicle, rodName, baitName, runId, not rodItem, not baitItem) then
        return false
    end

    if not travelToWithVehicle(vehicle, SHOP_POS, runId) then return false end

    local hardwareShop = workspace:FindFirstChild("ShopZone_Hardware")
    if not hardwareShop then return false end

    if not rodItem then
        local ok = pcall(function()
            Net.get("purchase_consumable", hardwareShop, rodName)
        end)
        if not ok then return false end
        task.wait(0.25)
    end

    if not baitItem then
        for _ = 1, baitAmount do
            if not isRunning(runId) then return false end
            pcall(function()
                Net.get("purchase_consumable", hardwareShop, baitName)
            end)
            task.wait(0.06)
        end
    end

    task.wait(0.4)
    return findInventoryItem(rodName) ~= nil and findInventoryItem(baitName) ~= nil
end

local function equipRod(rodItem)
    if not rodItem then return false end
    local player, character, humanoid = refreshCharacter()
    if not humanoid then return false end

    local equipped = rodItem:FindFirstChild("ItemEquipped")
    if equipped and equipped.Visible then
        return true
    end

    pcall(function()
        Net.get("toggle_equip_item", tostring(rodItem.Name))
    end)
    task.wait(0.15)

    local backpack = player:FindFirstChild("Backpack")
    local label = rodItem:FindFirstChild("ItemName")
    local toolName = label and label.Text or rodItem.Name
    local tool = character:FindFirstChild(toolName) or (backpack and backpack:FindFirstChild(toolName))

    if tool and tool:IsA("Tool") and tool.Parent == backpack then
        pcall(function() humanoid:EquipTool(tool) end)
        task.wait(0.2)
    end

    return character:FindFirstChild(toolName) ~= nil
end

local function getRodTool(rodItem)
    if not rodItem then return nil end
    local player, character = refreshCharacter()
    local label = rodItem:FindFirstChild("ItemName")
    local name = label and label.Text
    if not name then return nil end
    return character:FindFirstChild(name) or (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild(name))
end

-- 8) Fishing action
-- ============================================================
local function doFishingCycle(rodItem, baitItem, runId)
    if not rodItem or not baitItem then return false end
    if not equipRod(rodItem) then return false end

    local rodTool = getRodTool(rodItem)
    if not rodTool or not rodTool:IsA("Tool") then return false end

    local okAmmo = pcall(function()
        Net.get("equip_ammo_on_item", rodItem.Name, baitItem.Name)
    end)
    if not okAmmo then
        -- รองรับระบบที่ใช้ UID ของ inventory item
        okAmmo = pcall(function()
            Net.get("equip_ammo_on_item", tostring(rodItem.Name), tostring(baitItem.Name))
        end)
    end
    if not okAmmo then return false end
    task.wait(0.12)

    local okThrow = pcall(function()
        Net.get("throw_rod", rodTool, CAST_POS)
    end)
    if not okThrow then return false end

    -- รอ minigame แบบมี timeout ป้องกันค้าง
    local gui = getPlayerGui()
    local started = os.clock()
    local visible = false
    while isRunning(runId) and os.clock() - started < 20 do
        task.wait(0.05)
        gui = getPlayerGui()
        local slider = gui and gui:FindFirstChild("SliderMinigame")
        local frame = slider and slider:FindFirstChildOfClass("Frame")
        if frame and frame.Visible then
            visible = true
            break
        end
    end

    if not isRunning(runId) then return false end

    if visible then
        task.wait(0.05)
        pcall(function()
            Net.send("reel_ended", rodTool, true)
        end)
        task.wait(0.45)
        return true
    end

    pcall(function()
        Net.send("reel_ended", rodTool, false)
    end)
    pcall(function() refreshCharacter() end)
    task.wait(0.3)
    return false
end

-- 9) Sell fish - ใช้รถและใต้ดินเหมือนทุกจุดอื่น
-- ============================================================
local function sellAllFish(vehicle, runId)
    if CountFish() <= 0 then return true end

    local sellBeacon = workspace:FindFirstChild("SellFishBeacon")
    local touchPart = sellBeacon and sellBeacon:FindFirstChild("TouchPart")
    if not touchPart then return false end

    if not driveUnderground(vehicle, touchPart.Position, runId, 5, function()
        pcall(function() Sf:ExitVehicle(5) end)
    end) then
        return false
    end

    local fishes = {}
    local holding = getHolding()
    if holding then
        for _, item in ipairs(holding:GetChildren()) do
            if item:GetAttribute("ItemType") == "fish" then
                table.insert(fishes, item.Name)
            end
        end
    end

    -- เดินระยะสั้นเฉพาะจุด interaction ไม่ใช้การเดินทางเป็นเส้นทางหลัก
    for _, uid in ipairs(fishes) do
        if not isRunning(runId) then return false end
        pcall(function() Net.get("sell_item", tostring(uid), 1) end)
        task.wait(0.04)
    end
    task.wait(0.35)
    return true
end

-- 10) Auto Farm Fishing
-- ============================================================

func["AutoFarmFishing"] = function()
    fishingRunId += 1
    local runId = fishingRunId

    while isRunning(runId) do
        local vehicle = ensureFishingVehicle(runId)
        if not vehicle then
            task.wait(0.75)
            continue
        end

        local rodName = getBestFishingItem(FishingRods, Config.FishingRod)
        local baitName = getBestFishingItem(FishingBaits, Config.FishingWormtec)
        local rodItem = findInventoryItem(rodName)
        local baitItem = findInventoryItem(baitName)

        -- ถ้าไม่มีของ ให้ซื้อด้วยรถ ไม่ใช้ Teleport เดิน
        if not rodItem or not baitItem then
            if not buyMissingEquipment(vehicle, rodName, baitName, runId) then
                task.wait(0.6)
                continue
            end
            vehicle = ensureFishingVehicle(runId)
            if not vehicle then
                task.wait(0.6)
                continue
            end
        end

        -- ไปจุดตกปลาด้วยรถ + ใต้ดินเท่านั้น
        if not driveUnderground(vehicle, FISHING_SPOT, runId, 5, function()
            pcall(function() Sf:ExitVehicle(5) end)
            task.wait(0.2)
            local _, character = refreshCharacter()
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if root then
                root.CFrame = CFrame.new(FISHING_SPOT + Vector3.new(0, 0, -3), FISHING_SPOT)
            end
        end) then
            task.wait(0.5)
            continue
        end

        rodItem = findInventoryItem(rodName)
        baitItem = findInventoryItem(baitName)
        if rodItem and baitItem then
            doFishingCycle(rodItem, baitItem, runId)
        else
            task.wait(0.25)
        end

        -- เต็มแล้วขายทันที
        if CountFish() >= 15 then
            vehicle = ensureFishingVehicle(runId)
            if vehicle then
                sellAllFish(vehicle, runId)

                -- กลับไปซื้อเหยื่อด้วยรถ/ใต้ดิน
                if isRunning(runId) then
                    vehicle = ensureFishingVehicle(runId)
                    if vehicle and sitInFishingVehicle(vehicle, runId) then
                        driveUnderground(vehicle, SHOP_POS, runId, 5, function()
                            pcall(function() Sf:ExitVehicle(5) end)
                        end)

                        local hardwareShop = workspace:FindFirstChild("ShopZone_Hardware")
                        if hardwareShop then
                            local baitToBuy = getBestFishingItem(FishingBaits, Config.FishingWormtec)
                            local amount = math.clamp(math.floor(tonumber(Config.FishingBaitAmount) or 15), 1, 50)
                            for _ = 1, amount do
                                if not isRunning(runId) then break end
                                pcall(function()
                                    Net.get("purchase_consumable", hardwareShop, baitToBuy)
                                end)
                                task.wait(0.06)
                            end
                        end
                    end
                end
            end
        end

        task.wait(0.15)
    end

    pcall(function() Sf:ForceStop() end)
end

    Title = "🔢 Bait Purchase Amount",
    Step = 1,
    Value = {Min = 1, Max = 50, Default = Config.FishingBaitAmount},
    Callback = function(value)
        Config.FishingBaitAmount = math.clamp(math.floor(value), 1, 50)
    end,
})

local autoEquipActive = false
FishingTab:Toggle({
    Title = "☑️ Auto Equip FishingRod",
    Desc = "พยายามใส่คันเบ็ดจริงก่อนเริ่มแต่ละรอบ",
    Icon = "check",
    Type = "Checkbox",
    Value = false,
    Callback = function(value)
        autoEquipActive = value
        if value then
            task.spawn(function()
                while autoEquipActive do
                    local rodName = getBestFishingItem(FishingRods, Config.FishingRod)
                    local rod = findInventoryItem(rodName)
                    if rod then
                        pcall(function() equipRod(rod) end)
                    end
                    task.wait(1)
                end
            end)
        end
    end,
})

    Icon = "shield",
})
