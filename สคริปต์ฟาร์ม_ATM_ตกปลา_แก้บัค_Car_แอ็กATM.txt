local function GetNearThing(folder, path, condition)
    local closest, shortestDist = nil, math.huge
    if not folder then return nil end
    for _, obj in ipairs(folder:GetChildren()) do
        if string.find(obj.Name, path) then
            if not condition or condition(obj) then
                local dist = math.huge
                if obj:IsA("BasePart") then dist = distTo(obj.Position)
                elseif obj:IsA('Model') and obj.PrimaryPart then dist = distTo(obj.PrimaryPart.Position) end
                if dist < shortestDist then closest = obj shortestDist = dist end
            end
        end
    end
    return closest
end

local function GetDepositATMPos()
    if Config.AutoDepositJob then
        local cf = LOCKED_ATM_MAP[Config.SelectedJob]
        if cf then return cf.Position end
    end
    local shortestDist, atmPos = math.huge, nil
    for _, atm in ipairs(GetAllATM()) do
        local pos = atm.PrimaryPart and atm.PrimaryPart.Position
        if not pos then
            for _, part in ipairs(atm:GetDescendants()) do
                if part:IsA("BasePart") then pos = part.Position break end
            end
        end
        if pos then
            local d = distTo(pos)
            if d < shortestDist then shortestDist = d atmPos = pos end
        end
    end
    return atmPos

local pathParts = {}
function ClearPathLines()
    for _, part in ipairs(pathParts) do
        if part and part.Parent then part:Destroy() end
    end
    pathParts = {}
end
local function DrawPath(waypoints)
    ClearPathLines()
    if not waypoints or #waypoints < 2 then return end
    for i = 1, #waypoints - 1 do
        local p1 = waypoints[i].Position
        local p2 = waypoints[i+1].Position
        local dist = (p2 - p1).Magnitude
        if dist > 0.5 then
            local mid = (p1 + p2) / 2
            local dir = (p2 - p1).Unit
            local part = Instance.new("Part")
            part.Size = Vector3.new(0.4, 0.4, dist)
            part.CFrame = CFrame.lookAt(mid, mid + dir)
            part.Anchored = true
            part.CanCollide = false
            part.Material = Enum.Material.Neon
            part.BrickColor = BrickColor.new("Bright blue")
            part.Transparency = 0.25
            part.Parent = workspace
            table.insert(pathParts, part)
        end
    end
end

function Sf:GetInfo(w)
    local amount = 0
    local IsHaving = false
    local Uid = nil
    local Using = false
    local Drowning = false
    local Items = PlayerGui:FindFirstChild("Items")
    local holder = Items and Items:FindFirstChild("ItemsHolder")
    local Holding = holder and holder:FindFirstChild("ItemsScrollingFrame")
    if not Holding then return {0, false, nil, false, false} end
    for _, v in pairs(Holding:GetChildren()) do
        if v.Name ~= 'Folder' and v.Name ~= 'UIGridLayout' and v.Name ~= "ItemTemplate" then
            local itemName = v:FindFirstChild("ItemName")
            if itemName and itemName.Text == w then
                Uid = v.Name
                amount = amount + 1
                IsHaving = true
                local equipped = v:FindFirstChild("ItemEquipped")
                local destroyed = v:FindFirstChild("DestroyedItemIcon")
                Using = equipped and equipped.Visible or false
                Drowning = destroyed and destroyed.Visible or false
            end
        end
    end
    return {amount, IsHaving, Uid, Using, Drowning}
end

function Sf:IsVehicleType(vehicle, desiredType, expectedName)
    if not vehicle or not desiredType then return false end
    local wanted = tostring(desiredType):lower()
    local attrs = {
        vehicle:GetAttribute("VehicleType"),
        vehicle:GetAttribute("VehicleClass"),
        vehicle:GetAttribute("ItemType"),
        vehicle:GetAttribute("Type"),
        vehicle:GetAttribute("VehicleName"),
        vehicle:GetAttribute("ModelName")
    }
    local expected = expectedName and tostring(expectedName):lower() or nil
    local modelName = tostring(vehicle.Name):lower()
    if expected then
        local compactExpected = expected:gsub("[%s_%-]", "")
        local compactModel = modelName:gsub("[%s_%-]", "")
        if compactExpected ~= "" and (compactModel == compactExpected or compactModel:find(compactExpected, 1, true) or compactExpected:find(compactModel, 1, true)) then
            return true
        end
    end
    for _, attr in ipairs(attrs) do
        if attr then
            local a = tostring(attr):lower()
            if a == wanted or (wanted == "bike" and (a == "bicycle" or a == "motorbike" or a == "bmx")) then
                return true
            end
            if expected and a == expected then
                return true
            end
        end
    end
    if wanted == "bike" then
        return modelName:find("bike", 1, true) ~= nil or modelName:find("bmx", 1, true) ~= nil or modelName:find("bicycle", 1, true) ~= nil
    end
    -- รถยนต์บางคันไม่มีชื่อ/Attribute ว่า Car จึงห้ามพึ่งชื่ออย่างเดียว
    return modelName:find("car", 1, true) ~= nil or modelName:find("vehicle", 1, true) ~= nil
end

function Sf:Teleport(destination, value, t, speed)
    if not destination then return end
    c().StopWalking = false
    c()['Running'] = true
    local char = Client.Character or Client.CharacterAdded:Wait()
    local RootPart = char:WaitForChild("HumanoidRootPart")
    local Humanoid = char:WaitForChild("Humanoid")
    local maxAttempts = 3
    local attempt = 0
    local pathSuccess = false
    local waypoints = nil
    while attempt < maxAttempts and not pathSuccess and value do
        attempt = attempt + 1
        local path = PathfindingService:CreatePath({
            AgentCanJump = true,
            AgentJumpHeight = 2.5,
            AgentHeight = 8,
            AgentRadius = 2.5,
            AgentMaxSlope = 90,
            Costs = {BlockedNode = 50, DoorArea = 1}
        })
        local success = pcall(function()
            path:ComputeAsync(RootPart.Position, destination)
        end)
        if success and path.Status == Enum.PathStatus.Success then
            pathSuccess = true
            waypoints = path:GetWaypoints()
        else
            task.wait(0.2)
        end
    end
    if not pathSuccess or not waypoints then
        local delta = destination - RootPart.Position
        local dist = delta.Magnitude
        if dist < 0.05 then
            c()["Running"] = false
            ClearPathLines()
            return
        end
        local dir = delta.Unit
        local speed = speed or Config.WalkingSpeed
        local startTime = tick()
        local movedDist = 0
        while movedDist < dist and not c().StopWalking and value do
            task.wait()
            c()['Running'] = true
            if c().StopWalking or not value or Humanoid.Health <= 0 then
                c()['Running'] = false
                break
            end
            local elapsedTime = tick() - startTime
            movedDist = math.min(elapsedTime * speed, dist)
            local newPos = RootPart.Position + dir * movedDist
            if Humanoid.Sit then Humanoid.Sit = false end
            RootPart:PivotTo(CFrame.new(newPos))
            Sf:Ac("set_sprinting_1", true)
        end
        c()['Running'] = false
        ClearPathLines()
        return
    end
    DrawPath(waypoints)
    for _, wp in pairs(waypoints) do
        local offsetY = (wp.Action == Enum.PathWaypointAction.Jump) and 10 or 4
        local targetPos = wp.Position + Vector3.new(0, offsetY, 0)
        local startPos = RootPart.Position
        local dir = (targetPos - startPos).Unit
        local dist = (targetPos - startPos).Magnitude
        local movedDist = 0
        local speed = speed or Config.WalkingSpeed
        local startTime = tick()
        while movedDist < dist and not c().StopWalking and value do
            task.wait()
            c()['Running'] = true
            if c().StopWalking or not value or Humanoid.Health <= 0 then
                c()['Running'] = false
                break
            end
            local elapsedTime = tick() - startTime
            movedDist = math.min(elapsedTime * speed, dist)
            local newPos = startPos + dir * movedDist
            if Humanoid.Sit then Humanoid.Sit = false end
            RootPart:PivotTo(CFrame.new(newPos))
            Sf:Ac("set_sprinting_1", true)
        end
        c()['Running'] = false
        if not value then break end
    end
    c()['Running'] = false
    ClearPathLines()
end
local function shouldContinue(value)
    if value == nil then
        return c().AutoFarmATM
    end
    return value
end
function Sf:EnterVehicle(vehicle, timeout)
    if not vehicle or not vehicle.PrimaryPart then return false end
    timeout = timeout or 8
    local startTime = tick()
    while not Humanoid.Sit and tick() - startTime < timeout do
        if not vehicle or not vehicle.Parent or not vehicle.PrimaryPart then return false end
        local fired = false
        for _, prompt in ipairs(vehicle:GetDescendants()) do
            if prompt:IsA("ProximityPrompt") and prompt.Enabled then
                local ok = pcall(function() fireproximityprompt(prompt) end)
                fired = ok or fired
                task.wait(0.08)
                if Humanoid.Sit then break end
            end
        end
        if Humanoid.Sit then break end
        if not fired then task.wait(0.25) else task.wait(0.15) end
    end
    return Humanoid.Sit
end
function Sf:ExitVehicle(timeout)
    timeout = timeout or 4
    if not Humanoid.Sit then return true end
    Net.get("exit_seat")
    local startTime = tick()
    while Humanoid.Sit and tick() - startTime < timeout do
        task.wait(0.15)
    end
    return not Humanoid.Sit
end
function Sf:Drive(model, destination, value, t, stopDist, onArrive, customSpeed, customWheelieEnabled, customWheelieAngle, customWheelieOnlyBike, customVehicleType)
    if not model or not model.PrimaryPart then
        c()['Running'] = false
        c()["StopWalking"] = false
        return
    end
    c()["StopWalking"] = false
    local underground = (c().AutoFarmATM and c().ATMUnderground) or (c().AutoFarmFishing and c().FishingUnderground)
    local undergroundDepth = math.max(10, tonumber(c().FarmUndergroundDepth) or 55)
    if underground then
        local surfaceDestination = destination
        local travelDestination = destination + Vector3.new(0, -undergroundDepth, 0)
        local pivot = model:GetPivot()
        model:PivotTo(CFrame.new(pivot.Position.X, travelDestination.Y, pivot.Position.Z) * pivot.Rotation)
        task.wait(0.05)
        local startPos = model.PrimaryPart.Position
        local delta = travelDestination - startPos
        local dist = delta.Magnitude
        if dist > 0.05 then
            local dir = delta.Unit
            local speed = math.max(1, tonumber(customSpeed) or tonumber(c().DrivingSpeed) or 60)
            local started = tick()
            while shouldContinue(value) and model and model.PrimaryPart do
                task.wait()
                if not Humanoid.Sit and not Sf:EnterVehicle(model, 3) then break end
                local moved = math.min((tick() - started) * speed, dist)
                local newPos = startPos + dir * moved
                local rotation = CFrame.lookAt(newPos, newPos + dir)
                local wheelieEnabled = customWheelieEnabled
                if wheelieEnabled == nil then wheelieEnabled = c().WheelieEnabled end
                local wheelieOnlyBike = customWheelieOnlyBike
                if wheelieOnlyBike == nil then wheelieOnlyBike = c().WheelieOnlyBike end
                local wheelieAngle = tonumber(customWheelieAngle)
                if wheelieAngle == nil then wheelieAngle = tonumber(c().WheelieAngle) or 0 end
                local vehicleTypeForWheelie = customVehicleType or c().VechineType
                if wheelieEnabled and (not wheelieOnlyBike or vehicleTypeForWheelie == 'Bike') then
                    rotation = rotation * CFrame.Angles(math.rad(math.clamp(wheelieAngle, 0, 35)), 0, 0)
                end
                model:PivotTo(rotation)
                for _, part in pairs(model:GetDescendants()) do
                    if part:IsA('BasePart') then
                        part.AssemblyLinearVelocity = Vector3.zero
                        part.AssemblyAngularVelocity = Vector3.zero
                    end
                end
                if moved >= dist or (stopDist and (model.PrimaryPart.Position - travelDestination).Magnitude <= stopDist) then break end
            end
        end
        if model and model.PrimaryPart then
            local restorePos
            if shouldContinue(value) then
                restorePos = surfaceDestination
            else
                local p = model.PrimaryPart.Position
                restorePos = Vector3.new(p.X, surfaceDestination.Y, p.Z)
            end
            model:PivotTo(CFrame.new(restorePos) * model:GetPivot().Rotation)
            for _, part in pairs(model:GetDescendants()) do
                if part:IsA('BasePart') then
                    part.AssemblyLinearVelocity = Vector3.zero
                    part.AssemblyAngularVelocity = Vector3.zero
                end
            end
            if onArrive then pcall(onArrive) end
        end
        c().Running = false
        ClearPathLines()
        return true
    end
    local maxAttempts = 3
    local attempt = 0
    local pathSuccess = false
    local waypoints = nil
    while attempt < maxAttempts and not pathSuccess and shouldContinue(value) do
        attempt = attempt + 1
        local path = PathfindingService:CreatePath({
            AgentRadius = (customVehicleType or c().VechineType) == "Car" and 5 or 3,
            AgentHeight = 8,
            AgentCanJump = true,
            AgentMaxSlope = 50,
            AgentCanClimb = false,
            WaypointSpacing = 6,
            Costs = {
                BlockedNode = 100,
                Cars = 1,
                Water = math.huge,
                DoorArea = 1
            }
        })
        local success = pcall(function()
            path:ComputeAsync(model.PrimaryPart.Position, destination)
        end)
        if success and path.Status == Enum.PathStatus.Success then
            pathSuccess = true
            waypoints = path:GetWaypoints()
        else
            task.wait(0.2)
        end
    end
    if not pathSuccess or not waypoints then
        local startPos = model:GetPivot().Position
        local delta = destination - startPos
        local dist = delta.Magnitude
        if dist < 0.05 then
            c()["Running"] = false
            ClearPathLines()
            if onArrive then onArrive() end
            return
        end
        local dir = delta.Unit
        local speed = tonumber(customSpeed) or c().DrivingSpeed
        local startTime = tick()
        local movedDist = 0
        while movedDist < dist and shouldContinue(value) do
            task.wait(0)
            if not model or not model.PrimaryPart then break end
            if not Humanoid.Sit then
                if not Sf:EnterVehicle(model, 3) then break end
            end
            local dt = tick() - startTime
            movedDist = math.min(dt * speed, dist)
            -- ใช้ startPos เป็นฐานทุกเฟรม ไม่บวก movedDist ซ้ำจากตำแหน่งปัจจุบัน
            -- ป้องกันรถพุ่ง/เร็วเกินค่าที่ตั้งไว้ใน fallback movement
            local newPos = startPos + dir * movedDist
            local Rotation = CFrame.lookAt(newPos, newPos + dir)
            local wheelieEnabled = customWheelieEnabled
            if wheelieEnabled == nil then wheelieEnabled = c().WheelieEnabled end
            local wheelieOnlyBike = customWheelieOnlyBike
            if wheelieOnlyBike == nil then wheelieOnlyBike = c().WheelieOnlyBike end
            local wheelieAngle = tonumber(customWheelieAngle)
            if wheelieAngle == nil then wheelieAngle = tonumber(c().WheelieAngle) or 0 end
            local vehicleTypeForWheelie = customVehicleType or c().VechineType
            if wheelieEnabled and (not wheelieOnlyBike or vehicleTypeForWheelie == "Bike") then
                Rotation = Rotation * CFrame.Angles(math.rad(math.clamp(wheelieAngle, 0, 35)), 0, 0)
            end
            model:PivotTo(Rotation)
            for _, part in pairs(model:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.AssemblyLinearVelocity = Vector3.zero
                    part.AssemblyAngularVelocity = Vector3.zero
                end
            end
            if stopDist and (model.PrimaryPart.Position - destination).Magnitude <= stopDist then
                if onArrive then onArrive() end
                break
            end
        end
        c()['Running'] = false
        ClearPathLines()
        return
    end
    DrawPath(waypoints)
    for i, wp in pairs(waypoints) do
        if not model or not model.PrimaryPart then
            c()['Running'] = false
            return
        end
        if not Humanoid.Sit then
            if not Sf:EnterVehicle(model, 4) then
                c()['Running'] = false
                return
            end
        end
        local startPos = model:GetPivot().Position
        local heightOffset = 2
        if wp.Action == Enum.PathWaypointAction.Jump then heightOffset = 8 end
        local goalPos = wp.Position + Vector3.new(0, heightOffset, 0)
        local dir = (goalPos - startPos).Unit
        if dir.Magnitude < 0.001 then dir = Vector3.new(1, 0, 0) end
        local dist = (goalPos - startPos).Magnitude
        local movedDist = 0
        local speed = tonumber(customSpeed) or c().DrivingSpeed
        local startTime = tick()
        local reached = false
        while movedDist < dist and shouldContinue(value) do
            task.wait(0)
            if not model or not model.PrimaryPart then
                c()['Running'] = false
                return
            end
            c()['Running'] = true
            if not shouldContinue(value) then
                c()['Running'] = false
                break
            end
            if not Humanoid.Sit then
                if not Sf:EnterVehicle(model, 3) then
                    c()['Running'] = false
                    return
                end
            end
            local dt = tick() - startTime
            movedDist = math.min(dt * speed, dist)
            local newPos = startPos + dir * movedDist
            local Rotation = CFrame.lookAt(newPos, newPos + dir)
            local wheelieEnabled = customWheelieEnabled
            if wheelieEnabled == nil then wheelieEnabled = c().WheelieEnabled end
            local wheelieOnlyBike = customWheelieOnlyBike
            if wheelieOnlyBike == nil then wheelieOnlyBike = c().WheelieOnlyBike end
            local wheelieAngle = tonumber(customWheelieAngle)
            if wheelieAngle == nil then wheelieAngle = tonumber(c().WheelieAngle) or 0 end
            local vehicleTypeForWheelie = customVehicleType or c().VechineType
            if wheelieEnabled and (not wheelieOnlyBike or vehicleTypeForWheelie == "Bike") then
                Rotation = Rotation * CFrame.Angles(math.rad(math.clamp(wheelieAngle, 0, 35)), 0, 0)
            end
            model:PivotTo(Rotation)
            for _, part in pairs(model:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.AssemblyLinearVelocity = Vector3.zero
                    part.AssemblyAngularVelocity = Vector3.zero
                end
            end
            if stopDist and (model.PrimaryPart.Position - destination).Magnitude <= stopDist then
                reached = true
                break
            end
        end
        c()['Running'] = false
        if not shouldContinue(value) then break end
        if reached and onArrive then
            onArrive()
            break
        end
    end
    c()['Running'] = false
    ClearPathLines()
end

function StateController:StopJob()
    self.JobSystemActive = false
    Config.StartFarmJob = false
    if self.JobCoroutine then
        local s = coroutine.status(self.JobCoroutine)
        if s == "running" or s == "suspended" then pcall(task.cancel, self.JobCoroutine) end
        self.JobCoroutine = nil
    end
    Sf:ForceStop()
    ClearPathLines()
end

        self.JobCoroutine = task.spawn(function()

        end)
    end

local function GetBestFishingItem(list, requested)
    if requested and requested ~= "Smart Select" then
        return requested
    end
    local level = Sf:GetLevel()
    local best = nil
    local bestLevel = -1
    for _, item in ipairs(list) do
        if item.Level <= level and item.Level >= bestLevel then
            best = item.Name
            bestLevel = item.Level
        end
    end
    return best or list[#list].Name

                if PuddlesFolder then
                    local PuddleNear = GetNearThing(PuddlesFolder, "Puddle", function(puddle)

                        local OilPuddle = puddle.Name == "OilPuddle"
                        local Toxic = puddle.Name == "ToxicPuddle"
                        local Nasty = puddle.Name == "NastyPuddle"
                        return puddle:IsA("BasePart")
                            and puddle.Size.Magnitude > 0.001
                            and puddle.Transparency < 1
                            and (not Large or Level >= 5)
                            and (not OilPuddle or Level >= 35)
                            and (not Toxic or Level >= 50)
                            and (not Nasty or Level >= 35)
                    end)
                    if not CheckingIsProgess() and PuddleNear then
                        Sf:Teleport(PuddleNear.Position, Config.StartFarmJob)
                        RootPart:PivotTo(CFrame.new(RootPart.CFrame.Position) * CFrame.Angles(0, math.rad(160), 0))

                                end

local function ClearHighlights()
    for _, atm in pairs(GetAllATMForViewer()) do
        local hl = atm:FindFirstChild("ATMViewerHighlight")
        if hl then hl:Destroy() end
    end
end
local function IsATMAvailable(atm)
    local keys = {}
    for k in pairs(atm:GetAttributes()) do
        table.insert(keys, k)
    end
    table.sort(keys)
    if keys[3] and atm:GetAttribute(keys[3]) == false then
        return true
    end
    return false
end

local CachedBankLabel = nil

        local cleanText = CachedBankLabel.Text:gsub(",", "")

            CachedBankLabel = v

        end
    end
    return 0
end

local CharModule = require(game.ReplicatedStorage.Modules.Core.Char)
local enabled = false
local flickering = false
local undergroundBaseCFrame = nil
local DROP_DEPTH = -55
local MOVE_RADIUS = 10
local FLICKER_RATE = 0.1
local function isDowned()
    local hum = CharModule.get_hum()
    return hum and (hum:GetAttribute("HasBeenDowned") or hum:GetAttribute("IsDead") or hum.Health <= 0)
end
local function getHRP()
    local char = CharModule.current_char.get()
    if not char then return end
    return char:FindFirstChild("HumanoidRootPart")
end
local function teleportUnderground()
    local hrp = getHRP()
    if not hrp then return end
    local original = hrp.CFrame
    undergroundBaseCFrame = original + Vector3.new(0, DROP_DEPTH, 0)
    hrp.CFrame = undergroundBaseCFrame
end
local function flickerAndMove()
    if flickering then return end
    flickering = true
    task.spawn(function()
        while flickering and enabled and isDowned() do
            local hrp = getHRP()
            if hrp and undergroundBaseCFrame then
                local angle = math.random() * math.pi * 2
                local offset = Vector3.new(math.cos(angle), 0, math.sin(angle)) * MOVE_RADIUS
                local randomPos = undergroundBaseCFrame.Position + offset
                hrp.CFrame = CFrame.new(randomPos)
                task.wait(0.05)
                hrp.CFrame = undergroundBaseCFrame
            end
            task.wait(FLICKER_RATE)
        end
        flickering = false
    end)
end

PlayerTab:Toggle({
    Title = "📷 Third Person Camera",
    Icon = "camera",
    Type = "Checkbox",
    Value = false,
    Callback = function(v)
        Config.ThirdPerson = v
        if v then
            task.spawn(function()
                local cameraDistance = 25
                local minDistance = 5
                local maxDistance = 50
                local pitchAngle = -65
                local yawAngle = 0
                local SENSITIVITY = 0.3
                local isRightMouseDown = false
                UserInputService.InputBegan:Connect(function(input, gp)
                    if gp then return end
                    if input.UserInputType == Enum.UserInputType.MouseButton2 then
                        isRightMouseDown = true
                        UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
                    end
                end)
                UserInputService.InputEnded:Connect(function(input, gp)
                    if input.UserInputType == Enum.UserInputType.MouseButton2 then
                        isRightMouseDown = false
                        UserInputService.MouseBehavior = Enum.MouseBehavior.Default
                    end
                end)
                UserInputService.InputChanged:Connect(function(input, gp)
                    if gp then return end
                    if input.UserInputType == Enum.UserInputType.MouseMovement and isRightMouseDown then
                        yawAngle = yawAngle - input.Delta.X * SENSITIVITY
                        pitchAngle = math.clamp(pitchAngle - input.Delta.Y * SENSITIVITY, -89, -10)
                    end
                    if input.UserInputType == Enum.UserInputType.MouseWheel then
                        cameraDistance = math.clamp(cameraDistance - input.Position.Z * 2, minDistance, maxDistance)
                    end
                end)
                while Config.ThirdPerson do
                    task.wait()
                    local character = Client.Character
                    if not character then continue end
                    local hrp = character:FindFirstChild("HumanoidRootPart")
                    if not hrp then continue end
                    local camera = workspace.CurrentCamera
                    camera.CameraType = Enum.CameraType.Scriptable
                    local targetPosition = hrp.Position + Vector3.new(0, 1.5, 0)
                    local rotation = CFrame.Angles(0, math.rad(yawAngle), 0) * CFrame.Angles(math.rad(pitchAngle), 0, 0)
                    local cameraCFrame = CFrame.new(targetPosition) * rotation * CFrame.new(0, 0, cameraDistance)
                    camera.CFrame = cameraCFrame
                end
                workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
            end)
        else
            workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
        end
    end
})

    Desc = "แสดงคันเบ็ดติดตัว (visual)",
    Icon = "check",
    Type = "Checkbox",
    Value = false,
    Callback = function(v)
        autoEquipActive = v
        if not autoEquipActive then
            clearFakeTools()
        end
        if autoEquipActive then
            task.spawn(function()
                while autoEquipActive do
                    local player = game.Players.LocalPlayer
                    local character = player.Character or player.CharacterAdded:Wait()
                    local backpack = player:FindFirstChild("Backpack")
                    local humanoid = character:FindFirstChildOfClass("Humanoid")
                    local rArm = character:FindFirstChild("RightHand") or character:FindFirstChild("Right Arm")
                    local toolNames = {
                        "FishingRodUltimate",
                        "FishingRodAdvanced",
                        "FishingRodPro",
                        "FishingRodRegular"
                    }
                    if backpack and humanoid and rArm then
                        local mainTool = nil
                        for _, name in ipairs(toolNames) do
                            mainTool = character:FindFirstChild(name) or backpack:FindFirstChild(name)
                            if mainTool then break end
                        end
                        if mainTool and mainTool.Parent == backpack then
                            humanoid:EquipTool(mainTool)
                        end
                        for _, name in ipairs(toolNames) do
                            local tool = backpack:FindFirstChild(name)
                            if tool and tool ~= mainTool and not character:FindFirstChild(name .. "_Fake") then
                                local handle = tool:FindFirstChild("Handle")
                                if handle then
                                    local fakeHandle = handle:Clone()
                                    fakeHandle.Name = name .. "_Fake"
                                    fakeHandle.CanCollide = false
                                    fakeHandle.Parent = character
                                    local weld = Instance.new("WeldConstraint")
                                    weld.Part0 = rArm
                                    weld.Part1 = fakeHandle
                                    weld.Parent = fakeHandle
                                    fakeHandle.CFrame = rArm.CFrame
                                    table.insert(fakeTools, fakeHandle)
                                end
                            end
                        end
                    end
                    task.wait(1)
                end
            end)
        end
    end
})
local ConfigTab = Window:Tab({
    Title = '⚙️ Config',
    Icon = 'settings',
})
ConfigTab:Section({
    Title = '💾 Config Management',
    Icon = 'save'
})
local ConfigManager = WindUI.ConfigManager
local ConfigObj = ConfigManager and ConfigManager:CreateConfig("CathubConfig") or {
    Register = function() end,
    Save = function() end,
    Delete = function() end,
    Load = function() end,
}
local saveConfig = ConfigObj.Save
local deleteConfig = ConfigObj.Delete
local loadConfig = ConfigObj.Load
local saveBtn = ConfigTab:Button{
    Title = "💾 Save Config",
    Desc = "บันทึกการตั้งค่า",
    Icon = "save",
    Callback = function()
        if saveConfig then
            saveConfig(Config)
        end
    end
}
ConfigObj:Register("SaveConfig", saveBtn)
local deleteBtn = ConfigTab:Button{
    Title = "🗑️ Delete Config",
    Desc = "ลบบันทึกการตั้งค่า",
    Icon = "trash",
    Callback = function()
        if deleteConfig then
            deleteConfig(Config)
        end
    end
}
ConfigObj:Register("DeleteConfig", deleteBtn)
if loadConfig then
    loadConfig(Config)
end
