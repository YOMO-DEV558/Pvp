local function attempt_enable_desync()
    if Raknet then
        if Raknet.desync then
            Raknet.desync(true)
            return true
        end
    end
    
    if raknet then
        if raknet.desync then
            raknet.desync(true)
            return true
        end
    end
    
    return false
end

local function change_character_color(color)
    if not character_ref then
        return false
    end
    
    local descendants = character_ref:GetDescendants()
    
    for _, part in ipairs(descendants) do
        if part:IsA("BasePart") then
            part.Color = color
            local highlight = part:FindFirstChild("Highlight")
            if highlight then
                highlight.FillColor = color
            end
        end
    end
    
    return true
end

local function disable_desync()
    if Raknet and Raknet.desync then
        Raknet.desync(false)
    elseif raknet and raknet.desync then
        raknet.desync(false)
    elseif raknet and raknet.remove_send_hook then
        raknet.remove_send_hook()
    end
end

local is_desync_enabled = true

local function apply_color()
    if is_desync_enabled then
        change_character_color(Color3.fromRGB(0, 255, 100))
    else
        change_character_color(Color3.fromRGB(255, 50, 50))
    end
end

local function toggle_desync()
    is_desync_enabled = not is_desync_enabled
    
    if is_desync_enabled then
        local success = attempt_enable_desync()
        if not success then
            warn("⚠️ ไม่พบ RakNet library! ติดตั้ง Delta Executor จากโปรแกรม")
            warn("⚠️ RakNet not found! Install Delta Executor")
        end
    else
        disable_desync()
    end
    
    update_status_display()
    apply_color()
end
